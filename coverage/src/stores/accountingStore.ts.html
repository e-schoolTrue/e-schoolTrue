import { defineStore } from 'pinia'
import { ref } from 'vue'
import { ElMessage } from 'element-plus'
import { safeInvoke, strictInvoke, isProd } from '@/utils/ipc'

/** Ligne de courbe 30 jours. */
export interface CurvePoint {
  date: string
  entrees: number
  sorties: number
}

/** Indicateurs du tableau de bord comptable. */
export interface AccountingKpis {
  encaisseMois: number
  impayesTotal: number
  depensesMois: number
  soldeCaisse: number
  elevesAJour: number
  elevesEnRetard: number
}

/** Impayé élève. */
export interface ArrearRow {
  id: number
  firstname: string
  lastname: string
  matricule: string
  classe: string
  phone?: string
  totalDu: number
  totalPaye: number
  reste: number
  joursRetard: number
  statut: 'partiel' | 'impaye'
}

/** Dépense. */
export interface ExpenseRow {
  id: number
  date: string
  categorie: string
  libelle: string
  montant: number
  mode: string
  justificatif?: string
  acteur?: string
}

/** Mouvement de caisse. */
export interface CashMovement {
  id: number
  heure: string
  sens: 'entree' | 'sortie'
  motif: string
  montant: number
  mode: string
  reference?: string
}

/** État journalier de caisse. */
export interface CashDay {
  date: string
  ouvert: boolean
  fondOuverture: number
  totalEntrees: number
  totalSorties: number
  soldeTheorique: number
  soldeReel?: number
  ecart?: number
  cloture?: boolean
}

/** Heure enseignant pour la paie. */
export interface TeacherHourRow {
  id: number
  firstname: string
  lastname: string
  matiere: string
  heures: number
  tarifHoraire: number
  surcharge: number
  brut: number
  statut: 'brouillon' | 'valide' | 'paye'
}

function last30DaysLabels(): string[] {
  const out: string[] = []
  for (let i = 29; i >= 0; i--) {
    const d = new Date()
    d.setDate(d.getDate() - i)
    out.push(d.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit' }))
  }
  return out
}

const MOCK_CURVE_LABELS = last30DaysLabels()

/**
 * Store comptabilité avancée.
 * Chaque action tente d'abord le canal IPC contracté
 * (`comptabilite:*`, `expense:*`, `cash:*`, `bank:*`, `teacher:*`)
 * puis bascule sur un mock local si indisponible.
 */
export const useAccountingStore = defineStore('accounting', () => {
  const loading = ref(false)
  const kpis = ref<AccountingKpis>({

  const curveLabels = ref<string[]>([...MOCK_CURVE_LABELS])

  const curveSorties = ref<number[]>(MOCK_CURVE_LABELS.map((_, i) => 60_000 + ((i * 23_000) % 180_000)))

    values: [6_800_000, 3_100_000, 1_500_000, 1_050_000],
  })
  const arrears = ref<ArrearRow[]>([

  ])
  const expenses = ref<ExpenseRow[]>([
    { id: 1, date: new Date().toISOString().slice(0, 10), categorie: 'Fournitures', libelle: 'Rames de papier A4', montant: 450_000, mode: 'Espèces', acteur: 'Comptable' },
    { id: 2, date: new Date().toISOString().slice(0, 10), categorie: 'Maintenance', libelle: 'Réparation climatiseur', montant: 800_000, mode: 'Mobile Money', acteur: 'Intendant' },
  ])
  const cashDay = ref<CashDay>({
    date: new Date().toISOString().slice(0, 10),
    ouvert: true,
    fondOuverture: 500_000,
    totalEntrees: 1_250_000,
    totalSorties: 475_000,
    soldeTheorique: 1_275_000,
    cloture: false,
  })
  const movements = ref<CashMovement[]>([
    { id: 1, heure: '08:12', sens: 'entree', motif: 'Scolarité Mariam Diallo', montant: 300_000, mode: 'Espèces', reference: 'R-2026-0001' },
    { id: 2, heure: '10:05', sens: 'entree', motif: 'Inscription nouvel élève', montant: 950_000, mode: 'Mobile Money', reference: 'R-2026-0002' },
    { id: 3, heure: '11:40', sens: 'sortie', motif: 'Achat fournitures', montant: 475_000, mode: 'Espèces', reference: 'D-2026-0014' },
  ])
  const teacherHours = ref<TeacherHourRow[]>([
    { id: 1, firstname: 'Mamadou', lastname: 'Camara', matiere: 'Mathématiques', heures: 24, tarifHoraire: 50_000, surcharge: 0, brut: 1_200_000, statut: 'valide' },
    { id: 2, firstname: 'Fatoumata', lastname: 'Keïta', matiere: 'Français', heures: 20, tarifHoraire: 45_000, surcharge: 100_000, brut: 1_000_000, statut: 'brouillon' },
  ])

  /** Charge le tableau de bord (courbes + KPI + donuts). */
  async function fetchDashboard(): Promise<void> {
    loading.value = true
    try {
      const data = await safeInvoke<Partial<AccountingKpis> & { curve?: CurvePoint[] }>('comptabilite:dashboard', {})
      if (data && typeof data === 'object') {
        if (typeof (data as AccountingKpis).encaisseMois === 'number') {
          kpis.value = { ...kpis.value, ...(data as AccountingKpis) }
        }
        const curve = (data as { curve?: CurvePoint[] }).curve
        if (Array.isArray(curve) && curve.length) {
          curveLabels.value = curve.map((c) => c.date)
          curveEntrees.value = curve.map((c) => c.entrees)
          curveSorties.value = curve.map((c) => c.sorties)
        }
      }
      const rep = await safeInvoke<{ types?: { labels: string[]; values: number[] }; modes?: { labels: string[]; values: number[] } }>('comptabilite:repartition', {})
      if (rep?.types) donutTypes.value = rep.types
      if (rep?.modes) donutModes.value = rep.modes
    } finally {
      loading.value = false
    }
  }

  /** Charge les impayés. */
  async function fetchArrears(): Promise<void> {
    loading.value = true
    try {
      const data = await safeInvoke<ArrearRow[]>('comptabilite:impayes', [...arrears.value])
      if (Array.isArray(data) && data.length) arrears.value = data
    } finally {
      loading.value = false
    }
  }

  /** Charge les dépenses. */
  async function fetchExpenses(): Promise<void> {
    const data = await safeInvoke<ExpenseRow[]>('expense:list', [...expenses.value])
    if (Array.isArray(data) && data.length) expenses.value = data
  }

  /** Crée une dépense (avec justificatif déjà téléversé). Montant : pas de fallback silencieux en prod. */
  async function createExpense(payload: Omit<ExpenseRow, 'id'>): Promise<boolean> {
    try {
      const res = isProd
        ? await strictInvoke<{ id?: number }>('expense:create', payload)
        : await safeInvoke<{ id?: number }>('expense:create', { id: Date.now() }, payload)
      const next: ExpenseRow = { ...payload, id: res?.id ?? Date.now() }
      expenses.value = [next, ...expenses.value]
      cashDay.value.totalSorties += payload.montant
      cashDay.value.soldeTheorique = cashDay.value.fondOuverture + cashDay.value.totalEntrees - cashDay.value.totalSorties
      return true
    } catch (err) {
      if (!isProd) ElMessage.error('Échec de la création de la dépense')
      throw err
    }
  }

  /** Charge la journée de caisse. Montants : strict en prod. */
  async function fetchCashDay(dateISO: string): Promise<void> {
    if (isProd) {
      const data = await strictInvoke<CashDay>('cash:day', dateISO)
      cashDay.value = { ...data, date: dateISO }
      movements.value = await strictInvoke<CashMovement[]>('cash:movements', dateISO)
      return
    }
    const data = await safeInvoke<CashDay>('cash:day', { ...cashDay.value, date: dateISO }, dateISO)
    if (data) cashDay.value = { ...data, date: dateISO }
    const movs = await safeInvoke<CashMovement[]>('cash:movements', [...movements.value], dateISO)
    if (Array.isArray(movs)) movements.value = movs
  }

  /** Ouvre la caisse du jour. Montant : pas de fallback silencieux en prod. */
  async function openCash(fond: number): Promise<void> {
    if (isProd) await strictInvoke('cash:open', { fond })
    else await safeInvoke('cash:open', { ok: true }, { fond })
    cashDay.value = { ...cashDay.value, ouvert: true, fondOuverture: fond, cloture: false }
  }

  /** Clôture la caisse avec écart calculé. Montant : pas de fallback silencieux en prod. */
  async function closeCash(soldeReel: number): Promise<number> {
    if (isProd) await strictInvoke('cash:close', { soldeReel })
    else await safeInvoke('cash:close', { ok: true }, { soldeReel })
    const ecart = soldeReel - cashDay.value.soldeTheorique
    cashDay.value = { ...cashDay.value, soldeReel, ecart, cloture: true }
    return ecart
  }

  /** Ajoute un mouvement de caisse. Montant : pas de fallback silencieux en prod. */
  async function addMovement(payload: Omit<CashMovement, 'id'>): Promise<void> {
    if (isProd) await strictInvoke('cash:addMovement', payload)
    else await safeInvoke('cash:addMovement', { ok: true }, payload)
    const next: CashMovement = { ...payload, id: Date.now() }
    movements.value = [next, ...movements.value]
    if (payload.sens === 'entree') cashDay.value.totalEntrees += payload.montant
    else cashDay.value.totalSorties += payload.montant
    cashDay.value.soldeTheorique = cashDay.value.fondOuverture + cashDay.value.totalEntrees - cashDay.value.totalSorties
  }

  /** Charge les heures enseignants. */
  async function fetchTeacherHours(): Promise<void> {
    const data = await safeInvoke<TeacherHourRow[]>('teacher:hours', [...teacherHours.value])
    if (Array.isArray(data) && data.length) teacherHours.value = data
  }

  /** Valide / paie une ligne enseignant. Montant : pas de fallback silencieux en prod. */
  async function payTeacher(id: number): Promise<void> {
    if (isProd) await strictInvoke('teacher:pay', id)
    else await safeInvoke('teacher:pay', { ok: true }, id)
    teacherHours.value = teacherHours.value.map((t) => (t.id === id ? { ...t, statut: 'paye' } : t))
  }

  function validateTeacher(id: number): void {
    teacherHours.value = teacherHours.value.map((t) => (t.id === id ? { ...t, statut: 'valide' } : t))
  }

  return {
    loading,
    kpis,
    curveLabels,
    curveEntrees,
    curveSorties,
    donutTypes,
    donutModes,
    arrears,
    expenses,
    cashDay,
    movements,
    teacherHours,
    fetchDashboard,
    fetchArrears,
    fetchExpenses,
    createExpense,
    fetchCashDay,
    openCash,
    closeCash,
    addMovement,
    fetchTeacherHours,
    payTeacher,
    validateTeacher,
  }
})
