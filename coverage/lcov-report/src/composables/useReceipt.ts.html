import { ref } from 'vue'
import { ElMessage } from 'element-plus'
import { jsPDF } from 'jspdf'
import { amountInWordsFR } from '@/utils/amountInWordsFR'
import { safeInvoke, strictInvoke, isProd } from '@/utils/ipc'
import { formatCurrency } from '@/components/util/currencyFormatter'
import { useCurrency } from '@/composables/useCurrency'

export interface ReceiptData {
  id: string
  numero: string
  date: string
  eleve: string
  matricule: string
  classe: string
  montant: number
  mode: string
  motif: string
  caissier?: string
}

/**
 * Logique reçu : chargement école + reçu, impression, PDF, envoi.
 * Impression : `window.electronAPI.print` si dispo, sinon `window.print()`.
 * Montants (receipt:get / receipt:send) : strict en prod — aucun fallback silencieux.
 */
export function useReceipt(receiptId: string | number) {
  const loading = ref(false)
  const { currency } = useCurrency()
  const school = ref<{ name: string; address: string; phone: string; email: string; logo?: string }>({
    name: 'École',
    address: 'Non renseignée',
    phone: 'Non renseigné',
    email: 'Non renseigné',
  })
  const receipt = ref<ReceiptData>({
    id: String(receiptId ?? '1'),
    numero: 'R-2026-0001',
    date: new Date().toLocaleDateString('fr-FR'),
    eleve: 'Mariam Diallo',
    matricule: 'EL-2026-001',
    classe: '6ème A',
    montant: 300_000,
    mode: 'Espèces',
    motif: 'Frais de scolarité — Tranche 2',
    caissier: 'Comptable',
  })

  const montantLettres = ref('')

  async function load(): Promise<void> {
    loading.value = true
    try {
      const schoolData = await safeInvoke<Record<string, string>>('school:get', {})
      if (schoolData && typeof schoolData === 'object') {
        school.value = {
          name: String(schoolData.name ?? school.value.name),
          address: String(schoolData.address ?? school.value.address),
          phone: String(schoolData.phone ?? school.value.phone),
          email: String(schoolData.email ?? school.value.email),
        }
        const logoId = (schoolData as unknown as { logo?: { id?: number } }).logo?.id
        if (logoId) {
          try {
            const logoRes = await safeInvoke<{ type: string; content: string } | null>('school:getLogo', null, logoId)
            if (logoRes?.content) school.value.logo = `data:${logoRes.type};base64,${logoRes.content}`
          } catch {
            /* logo optionnel */
          }
        }
      }
      if (isProd) {
        const r = await strictInvoke<Partial<ReceiptData>>('comptabilite:receipt:get', receiptId)
        if (r && Object.keys(r).length) receipt.value = { ...receipt.value, ...r } as ReceiptData
      } else {
        const r = await safeInvoke<Partial<ReceiptData>>('comptabilite:receipt:get', {}, receiptId)
        if (r && Object.keys(r).length) receipt.value = { ...receipt.value, ...r } as ReceiptData
      }
      montantLettres.value = amountInWordsFR(receipt.value.montant)
    } finally {
      loading.value = false
    }
  }

  function printReceipt(): void {
    try {
      const w = window as unknown as { electronAPI?: { print?: (o: unknown) => Promise<unknown> } }
      if (w.electronAPI?.print) {
        void w.electronAPI.print({ title: `Reçu ${receipt.value.numero}` }).catch(() => window.print())
      } else {
        window.print()
      }
    } catch {
      window.print()
    }
  }

  function exportPdf(): void {
    const doc = new jsPDF()
    doc.setFont('helvetica', 'bold')
    doc.text(String(school.value.name), 14, 18)
    doc.setFont('helvetica', 'normal')
    doc.text(`Reçu ${receipt.value.numero}`, 14, 28)
    doc.text(`Date : ${receipt.value.date}`, 14, 36)
    doc.text(`Élève : ${receipt.value.eleve} (${receipt.value.matricule})`, 14, 44)
    doc.text(`Classe : ${receipt.value.classe}`, 14, 52)
    doc.text(`Motif : ${receipt.value.motif}`, 14, 60)
    doc.text(`Montant : ${formatCurrency(receipt.value.montant, currency.value)}`, 14, 68)
    doc.text(`En lettres : ${amountInWordsFR(receipt.value.montant)}`, 14, 76)
    doc.text(`Mode : ${receipt.value.mode}`, 14, 84)
    doc.save(`recu_${receipt.value.numero}.pdf`)
    ElMessage.success('Reçu exporté en PDF')
  }

  async function sendReceipt(): Promise<void> {
    if (isProd) {
      await strictInvoke('comptabilite:receipt:send', receipt.value.id)
    } else {
      await safeInvoke('comptabilite:receipt:send', { ok: true }, receipt.value.id)
    }
    ElMessage.success('Reçu envoyé avec succès')
  }

  return { loading, school, receipt, montantLettres, load, printReceipt, exportPdf, sendReceipt }
}
